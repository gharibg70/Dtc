  <script src="config.js"></script>
  <script>
    const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

    async function testDatabase() {
      const resultDiv = document.getElementById('result');
      resultDiv.innerHTML = '<div class="loading">جاري الاختبار...</div>';

      try {
        // قائمة الجداول المتوقعة
        const expectedTables = [
          'users',
          'buses',
          'routes',
          'stops',
          'schedules',
          'bus_tracking',
          'subscriptions',
          'notifications'
        ];

        let html = '';
        let successCount = 0;
        let errorCount = 0;
        const foundTables = [];

        // اختبار كل جدول
        for (const table of expectedTables) {
          try {
            const { data, error, count } = await supabase
              .from(table)
              .select('*', { count: 'exact', head: true });

            if (error) {
              if (error.message.includes('does not exist') || error.code === '42P01') {
                html += `<div class="table-item" style="border-right-color: #dc2626;">
                  <div class="table-name" style="color: #dc2626;">❌ ${table}</div>
                  <div class="table-count">الجدول غير موجود</div>
                </div>`;
                errorCount++;
              } else {
                html += `<div class="table-item" style="border-right-color: #f59e0b;">
                  <div class="table-name" style="color: #f59e0b;">⚠️ ${table}</div>
                  <div class="table-count">خطأ: ${error.message}</div>
                </div>`;
                errorCount++;
              }
            } else {
              html += `<div class="table-item">
                <div class="table-name">✅ ${table}</div>
                <div class="table-count">عدد السجلات: ${count || 0}</div>
              </div>`;
              foundTables.push(table);
              successCount++;
            }
          } catch (err) {
            html += `<div class="table-item" style="border-right-color: #dc2626;">
              <div class="table-name" style="color: #dc2626;">❌ ${table}</div>
              <div class="table-count">خطأ في الاتصال</div>
            </div>`;
            errorCount++;
          }
        }

        // عرض النتيجة النهائية
        let statusHtml = '';
        if (successCount === expectedTables.length) {
          statusHtml = `<div class="status success">
            ✅ ممتاز! جميع الجداول موجودة ومتصلة بنجاح (${successCount}/${expectedTables.length})
          </div>`;
        } else if (successCount > 0) {
          statusHtml = `<div class="status info">
            ⚠️ بعض الجداول موجودة (${successCount}/${expectedTables.length}) - تحتاج إلى إنشاء الجداول المفقودة
          </div>`;
        } else {
          statusHtml = `<div class="status error">
            ❌ لا توجد جداول - قاعدة البيانات فارغة أو غير موجودة
          </div>`;
        }

        resultDiv.innerHTML = statusHtml + '<div class="table-list">' + html + '</div>';

      } catch (err) {
        resultDiv.innerHTML = `<div class="status error">
          ❌ خطأ في الاتصال بقاعدة البيانات: ${err.message}
        </div>`;
      }
    }

    // تشغيل الاختبار عند تحميل الصفحة
    window.onload = testDatabase;
  </script>
